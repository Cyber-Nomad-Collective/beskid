## Application host (`app` target)

```beskid
host AppHost(string[] args) : ConsoleHost {
    registry {
        single AppConfig for IConfiguration;
        single SqlStorage for IStorage;
        single FileStorage for IStorage;
    }

    startup(IConfiguration config, IStorage[] storages) {
        for s in storages { s.Open(config); }
    }
}

type StorageAggregator {
    inject IConfiguration configuration;
    inject IStorage[] storages;
}

fn main() {
    launch AppHost(args);
}
```

## Library host composed by app

**`Lib` project** (no `launch`):

```beskid
host InfraHost : ConsoleHost {
    registry {
        single SharedLogger for ILogger;
    }
}
```

**`App` project:**

```beskid
host AppHost(string[] args) : InfraHost {
    registry {
        single AppLogger for ILogger;   // overrides InfraHost ILogger
    }

    startup(ILogger logger) { }
}

fn main() {
    launch AppHost(args);
}
```

## Named scope with `init`, `dispose`, and navigation

```beskid
host AppHost(string[] args) : ConsoleHost {
    registry {
        single Configuration for IConfiguration;
    }

    scope HttpScope(RequestContext request) {
        DbSession for IDbSession;

        init(global::IConfiguration config, IDbSession db) {
            db.Connect(config, request);
        }

        dispose(IDbSession db) {
            db.Close();
        }
    }

    startup(IConfiguration config) { }
}

type RequestWorker {
    inject global::IConfiguration configuration;

    fn Run(RequestContext request) {
        with HttpScope(request) {
            Process();
        }
    }
}
```

## Plural inject vs singular error

```beskid
registry {
    single A for IProvider;
    single B for IProvider;
}

type Consumer {
    inject IProvider[] providers;   // OK — both A and B
    // inject IProvider p;          // E1705 — use array form
}
```

## Nested scopes and independent activations

```beskid
with HttpScope(request) {
    with UnitOfWork(readOnly: false) { Save(); }
}
with HttpScope(request) {
    with UnitOfWork(readOnly: true) { Query(); }   // second activation — new scoped set
}
```
